import java.util.Scanner;
public class Main{
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int arr[][] = new int[n][n];
		for(int i=0;i<n;i++){
		    for(int j=0;j<n;j++){
		        arr[i][j] = sc.nextInt();
		    }
		}
	    int result = absdiff(n,arr);
	    System.out.print(result);
	}
	static int absdiff(int n,int arr[][]){
	    int pd = 0;
	    int sd = 0;
	    for(int i=0;i<n;i++){
	        pd = pd+arr[i][i];
	        sd = sd+arr[i][n-1-i];
	    }
	    int diff = pd-sd;
	    if(diff<0){
	        diff = -diff;
	    }
	    return diff;
	}
}
