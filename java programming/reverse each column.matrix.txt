import java.util.Scanner;
public class Main{
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int r = sc.nextInt();
		int c = sc.nextInt();
		int arr[][] = new int[r][c];
		for(int i=0;i<r;i++){
		    for(int j=0;j<c;j++){
		        arr[i][j] = sc.nextInt();
		    }
		}
		print(r,c,arr);
	}
	static void reversecol(int r,int c,int arr[][]){
	    for(int j=0;j<c;j++){
	        int s = 0;
	        int e = r-1;
	        while(s<e){
	            int t = arr[s][j];
	            arr[s][j] = arr[e][j];
	            arr[e][j] = t;
	            s++;
	            e--;
	        }
	    }
	}
	static void print(int r,int c,int arr[][]){
	    reversecol(r,c,arr);
	    for(int i=0;i<r;i++){
	        for(int j=0;j<c;j++){
	            System.out.print(arr[i][j]+" ");
	        }
	        System.out.println();
	    }
	}
}
