import java.util.Scanner;
public class Main{
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int r = sc.nextInt();
		int c = sc.nextInt();
		int arr[][] = new int[r][c];
		for(int i=0;i<r;i++){
		    for(int j=0;j<c;j++){
		        arr[i][j] = sc.nextInt();
		    }
		}
	    int result = sumarray(r,c,arr);
	    System.out.print(result);
	}
	static int sumarray(int r,int c,int arr[][]){
	    int sum = 0;
	    for(int i=0;i<r;i++){
	        for(int j=0;j<c;j++){
	            sum = sum + arr[i][j];
	        }
	    }
	    return sum;
	}
}
